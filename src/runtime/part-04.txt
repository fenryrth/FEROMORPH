  function randomSeed() {
    const values = new Uint32Array(1);
    if (window.crypto?.getRandomValues) {
      window.crypto.getRandomValues(values);
      return 1000 + (values[0] % 900000);
    }
    return 1000 + Math.floor(Math.random() * 900000);
  }

  function randomRange(min, max) {
    return min + Math.random() * (max - min);
  }

  function formatSeed(seed) {
    return String(Math.floor(seed)).padStart(6, "0");
  }

  function smoothstep(edge0, edge1, value) {
    const x = Math.min(1, Math.max(0, (value - edge0) / (edge1 - edge0)));
    return x * x * (3 - 2 * x);
  }

  function easeInOut(value) {
    return value < 0.5
      ? 4 * value * value * value
      : 1 - Math.pow(-2 * value + 2, 3) / 2;
  }

  function resize() {
    const width = Math.max(1, Math.floor(window.innerWidth * adaptiveScale));
    const height = Math.max(1, Math.floor(window.innerHeight * adaptiveScale));
    if (canvas.width !== width || canvas.height !== height) {
      canvas.width = width;
      canvas.height = height;
      gl.viewport(0, 0, width, height);
    }
  }

  function showHudTemporarily(duration = 3200) {
    window.clearTimeout(uiTimeout);
    hudVisible = true;
    hud.classList.add("is-visible");
    artwork.classList.add("ui-active");
    hud.setAttribute("aria-hidden", "false");

    uiTimeout = window.setTimeout(() => {
      hudVisible = false;
      hud.classList.remove("is-visible");
      artwork.classList.remove("ui-active");
      hud.setAttribute("aria-hidden", "true");
    }, duration);
  }

  function toggleHud() {
    window.clearTimeout(uiTimeout);
    hudVisible = !hudVisible;
    hud.classList.toggle("is-visible", hudVisible);
    artwork.classList.toggle("ui-active", hudVisible);
    hud.setAttribute("aria-hidden", String(!hudVisible));
  }
